ШАГ 1 — Что есть в CRM сейчас:
SELECT 
  s.name, s.client_id,
  DATE(o.created_at) as date,
  COUNT(DISTINCT o.id) as orders,
  SUM(oi.quantity) as qty,
  ROUND(SUM(oi.price * oi.quantity)::numeric, 0) as total
FROM orders o
JOIN stores s ON o.store_id = s.id
LEFT JOIN order_items oi ON o.id = oi.order_id
WHERE s.client_id IN ('3364383','3835567','2311038','2496152','4052691')
AND DATE(o.created_at) IN ('2026-03-19','2026-03-20')
AND o.status != 'cancelled'
AND (o.ozon_status IS NULL OR o.ozon_status != 'cancelled')
GROUP BY s.name, s.client_id, DATE(o.created_at)
ORDER BY s.client_id, date;

ШАГ 2 — Ключевой вопрос: что отличает Sunshine от остальных?
SELECT 
  s.name, s.client_id,
  COUNT(DISTINCT o.fulfillment_type) as fulfillment_types,
  STRING_AGG(DISTINCT o.fulfillment_type, ', ') as types
FROM orders o
JOIN stores s ON o.store_id = s.id
WHERE s.client_id IN ('3364383','3835567','2311038','2496152','4052691')
GROUP BY s.name, s.client_id;
