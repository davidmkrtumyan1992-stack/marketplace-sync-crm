FROM wb_supplies ws
WHERE ws.status = 'open'
AND EXISTS (
  SELECT 1 FROM orders o
  WHERE o.wb_supply_id = ws.supply_id
  AND o.source = 'wildberries'
  AND o.wb_status IN ('new', 'waiting', 'confirm')
);

// Шаг A — обновить заказы внутри устаревших поставок:

      status = 'completed'

    WHERE status = 'open'

  )
`);

// Шаг Б — закрыть устаревшие поставки:
await db.execute(sql`
  UPDATE wb_supplies
  SET status = 'closed', closed_at = NOW()
  WHERE status = 'open'
  AND created_at < NOW() - INTERVAL '21 days'
  AND organization_id = ${orgId}
  AND EXISTS (
    SELECT 1 FROM orders o
    WHERE o.wb_supply_id = wb_supplies.supply_id
    AND o.wb_status = 'new'
  )
`);

console.log('[wb-stale-sync] Устаревшие поставки (>21 дня) закрыты');

Важно: Шаг А выполнять ПЕРЕД Шагом Б —
сначала обновить статусы заказов потом закрыть поставки.
Иначе заказы останутся как незавершённые в статистике.
