SELECT ms.id, ms.marketplace, ms.store_id, ms.is_active,
  s.name as store_name
FROM marketplace_settings ms
LEFT JOIN stores s ON ms.store_id = s.id
WHERE s.id IS NULL;

SELECT ws.supply_id, ws.status, ws.store_id
FROM wb_supplies ws
LEFT JOIN stores s ON ws.store_id = s.id
WHERE (ws.store_id NOT IN (SELECT id FROM stores)
   OR ws.store_id IS NULL)
AND ws.status = 'open';

SELECT o.source, COUNT(*) as orphan_orders
FROM orders o
LEFT JOIN stores s ON o.store_id = s.id
WHERE s.id IS NULL

ШАГ 3 — Исправить эндпоинт DELETE /api/stores/:id:

Найди эндпоинт удаления магазина в server/routes.ts.
Заменить на новую логику:

app.delete("/api/stores/:id", isAuthenticated,
  requireRole("owner"), async (req, res) => {
  const storeId = Number(req.params.id);
  const orgId = getOrgId(req);

  const store = await storage.getStore(storeId);
  if (!store) return res.status(404).json({ message: "Магазин не найден" });

  // 1 — ПЕРВЫМ удалить API ключи:
  await db.delete(marketplaceSettingsTable)
    .where(eq(marketplaceSettingsTable.storeId, storeId));
  console.log(`[store-delete] API ключи магазина ${storeId} удалены`);

  // 2 — Закрыть открытые поставки WB:
  await db.update(wbSuppliesTable)
    .set({ status: 'closed', closedAt: new Date() })
    .where(and(
      eq(wbSuppliesTable.storeId, storeId),
      eq(wbSuppliesTable.status, 'open')
    ));

  // 3 — Удалить product_marketplace_links:
  await db.delete(productMarketplaceLinksTable)
    .where(eq(productMarketplaceLinksTable.storeId, storeId));

  // 4 — Удалить сам магазин (CASCADE удалит orders):
  await storage.deleteStore(storeId);

  console.log(`[store-delete] Магазин ${store.name} (${storeId}) полностью удалён`);
  return res.json({
    success: true,
    message: `Магазин «${store.name}» и все связанные данные удалены`
  });
});
