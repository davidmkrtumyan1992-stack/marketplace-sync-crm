ШАГ 2 — Проверь FBO vs FBS для каждого магазина за 20 марта:
SELECT 
  s.name, s.client_id,
  o.fulfillment_type,
  COUNT(*) as cnt,
  SUM(oi.quantity) as qty,
  ROUND(SUM(o.total_amount)::numeric, 0) as total
FROM orders o
JOIN stores s ON o.store_id = s.id
LEFT JOIN order_items oi ON o.id = oi.order_id
WHERE s.client_id IN ('3364383','3835567','2311038','2496152','4052691')
AND DATE(o.created_at AT TIME ZONE 'UTC' AT TIME ZONE 'Europe/Moscow') = '2026-03-20'
AND o.status != 'cancelled'
AND (o.ozon_status IS NULL OR o.ozon_status != 'cancelled')
GROUP BY s.name, s.client_id, o.fulfillment_type
ORDER BY s.client_id;
